# Leak-Free CV, Stacking & Calibration Pipeline Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Eliminate validation target leakage, in-sample calibration evaluation, greedy OOF hill-climbing, test pseudo-label sharpening, and arbitrary prevalence scaling across Stages 2–5, establishing a genuinely leak-free modeling pipeline.

**Architecture:** Invert Stage 2 fold loop to execute `screen_features()` strictly on outer-training splits with resilient categorical metadata handling; delegate `blend_and_calibrate()` to cross-fitted `platt_scaling_calibrate()`; replace Stage 5 greedy hill-climbing with $L_2$-regularized logit stacking; remove multiplicative test prevalence hacking; verify generalization on an untouched 20% shadow holdout.

**Tech Stack:** Python 3.10+, LightGBM, CatBoost, XGBoost, Scikit-learn, SciPy, Pandas, NumPy, Pytest.

## Global Constraints

- Competition metric definition in `src/metrics.py` ($0.40 \cdot \text{AUC} + 0.60 \cdot (1 - \text{LogLoss} / 0.595)$) MUST NOT be altered.
- All probability outputs must remain bounded within $[0.0020, 0.9980]$ (`FLOOR`, `CEIL`).
- All feature selection and scaling must observe ONLY outer training partitions ($X_{tr}, y_{tr}$).
- Calibrated out-of-fold predictions must strictly be generated by models that did not see those validation samples during calibration fitting.
- Test predictions must be computed as the fold-ensemble average of the fold models.

```python
import pytest
from src.ensemble.stacking import blend_and_calibrate

def test_blend_and_calibrate_is_cross_fitted():
    np.random.seed(42)
    n_samples = 200
    y_true = np.random.binomial(1, 0.15, size=n_samples)

    # Synthetic predictions from two models with slight noise
    m1 = np.clip(y_true * 0.7 + np.random.normal(0, 0.1, size=n_samples), 0.01, 0.99)
    m2 = np.clip(y_true * 0.6 + np.random.normal(0, 0.15, size=n_samples), 0.01, 0.99)

    oof_dict = {"m1": m1, "m2": m2}
    test_dict = {"m1": m1[:50], "m2": m2[:50]}

    weights, oof_b, test_b, oof_cal, test_cal = blend_and_calibrate(
        oof_dict, test_dict, y_true, n_splits=5, seed=42
    )

    # 1. Output shapes must match inputs
    assert oof_cal.shape == (n_samples,)
    assert test_cal.shape == (50,)

    # 2. Probability bounds
    assert np.all(oof_cal >= 0.0020) and np.all(oof_cal <= 0.9980)
    assert np.all(test_cal >= 0.0020) and np.all(test_cal <= 0.9980)

    # 3. Verify it is not an in-sample fit:
    # An in-sample logistic regression fitted on (z_oof, y_true) has a specific objective value.
    # Out-of-fold calibration evaluated across 5 folds must differ from a single global in-sample fit.
    from scipy.special import logit
    from sklearn.linear_model import LogisticRegression
    z_oof = logit(np.clip(oof_b, 1e-6, 1.0 - 1e-6)).reshape(-1, 1)
    in_sample_cal = LogisticRegression(C=1.0, solver="lbfgs").fit(z_oof, y_true)
    in_sample_preds = in_sample_cal.predict_proba(z_oof)[:, 1]

    # Out-of-fold cross-fitted predictions should not exactly match global in-sample fit
    assert not np.allclose(oof_cal, in_sample_preds, atol=1e-5)
```

```python
# experiments/run_shadow_holdout_benchmark.py
"""
Shadow Holdout Benchmark:
Evaluates the clean leak-free pipeline against an immutable 20% shadow holdout (8,000 samples)
to prove that the clean pipeline generalizes better than the overfitted pipeline.
"""
import os
import numpy as np
import pandas as pd
from sklearn.model_selection import StratifiedKFold
from src.config import TARGET, ID_COL, SEED, get_default_dataset_paths
from src.metrics import competition_score


def run_benchmark():
    train_path, test_path = get_default_dataset_paths()
    df = pd.read_csv(train_path)
    target_col = TARGET if TARGET in df.columns else "Target"
    y = df[target_col].to_numpy(int)

    # Immutable 80/20 shadow split
    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=SEED)
    dev_idx, shadow_idx = next(skf.split(df, y))

    dev_df = df.iloc[dev_idx].reset_index(drop=True)
    shadow_df = df.iloc[shadow_idx].reset_index(drop=True)
    y_shadow = shadow_df[target_col].to_numpy(int)

    print("=" * 80)
    print(f"SHADOW HOLDOUT BENCHMARK (Dev: {len(dev_df)} rows, Shadow: {len(shadow_df)} rows)")
    print(f"Shadow Target Prevalence: {y_shadow.mean():.5f}")
    print("=" * 80)


if __name__ == "__main__":
    run_benchmark()
```
